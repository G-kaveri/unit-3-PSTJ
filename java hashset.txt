
import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        Set<String> pairs = new HashSet<>();

        for (int i = 0; i < t; i++) {
            String a = sc.next();
            String b = sc.next();

            pairs.add(a + " " + b);

            System.out.println(pairs.size());
        }

        sc.close();
    }
}
